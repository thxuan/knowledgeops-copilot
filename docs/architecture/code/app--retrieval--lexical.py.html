22def bm25(query, hits, top_k, k1=1.2, b=0.75):
23 """Statistics use only the caller's authorized corpus."""
24 if not hits:
25 return []
26 counts = [Counter(tokenize(evidence_text(h))) for h in hits]
27 lengths = [sum(c.values()) for c in counts]
28 avg = sum(lengths) / len(lengths) or 1
29 frequency = Counter(t for c in counts for t in c)
30 terms = set(tokenize(query))
31 ranked = []
32 for hit, doc_counts, length in zip(hits, counts, lengths, strict=True):
33 score = 0.0
34 for term in terms:
35 tf = doc_counts[term]
36 if tf:
37 df = frequency[term]
38 idf = math.log1p((len(hits) - df + .5) / (df + .5))
39 score += idf * tf * (k1 + 1) / (tf + k1 * (1 - b + b * length / avg))
40 if score > 0:
41 ranked.append({**hit, "score": score})
42 return sorted(ranked, key=lambda h: (-h["score"], h["chunk_id"]))[:top_k]

45def rrf(rankings, rank_constant=60):
46 """Combine ranks, never add BM25 and cosine scores directly."""
47 if rank_constant < 1:
48 raise ValueError("RRF constant must be positive")
49 fused = {}
50 for name, hits in rankings.items():
51 seen = set()
52 for rank, hit in enumerate(hits, 1):
53 key = hit["chunk_id"]
54 if key in seen:
55 continue
56 seen.add(key)
57 if key not in fused:
58 fused[key] = {**hit, "score": 0.0, "ranks": {}, "component_scores": {}}
59 fused[key]["score"] += 1 / (rank_constant + rank)
60 fused[key]["ranks"][name] = rank
61 fused[key]["component_scores"][name] = hit["score"]
62 return sorted(fused.values(), key=lambda h: (-h["score"], h["chunk_id"]))
