46def test_rrf_discards_score_scales_and_deduplicates():
47 a, b = hit('a', score=.9), hit('b', score=.1)
48 keyword = [hit('b', score=1e6), hit('a', score=1e-3)]
49 fused = rrf({'vector': [a, b], 'keyword': keyword})
50 assert fused[0]['score'] == pytest.approx(1/61 + 1/62)
51 assert fused[0]['score'] == fused[1]['score']
