62 def submit(self, *, identity: Identity, filename: str, content: bytes, file_hash: str,
63 parser_version: str, source: str, version: str | None) -> dict:
64 now = time.time()
65 with self.connection() as db:
66 # Serialize the same upload key; unrelated uploads proceed independently.
67 db.execute("SELECT pg_advisory_xact_lock(hashtextextended(%s, 0))",
68 (json.dumps([identity.tenant_id, file_hash, parser_version]),))
69 document_id, task_id = str(uuid4()), str(uuid4())
70 cursor = db.execute("""
71 INSERT INTO documents(document_id, tenant_id, filename, size_bytes, file_hash,
72 parser_version, source, version, uploaded_by, content, created_at)
73 VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
74 ON CONFLICT(tenant_id, file_hash, parser_version) DO NOTHING
75 """, (document_id, identity.tenant_id, filename, len(content), file_hash,
76 parser_version, source, version, identity.user_id, content, now))
77 duplicate = cursor.rowcount == 0
78 if duplicate:
79 document_id = db.execute("""
80 SELECT document_id FROM documents
81 WHERE tenant_id = %s AND file_hash = %s AND parser_version = %s
82 """, (identity.tenant_id, file_hash, parser_version)).fetchone()["document_id"]
83 existing = db.execute("""SELECT d.active FROM documents d
84 JOIN document_access a USING(document_id)
85 WHERE d.document_id=%s AND a.user_id=%s FOR SHARE OF d, a""",
86 (document_id, identity.user_id)).fetchone()
87 if existing is None or not existing["active"]:
88 raise AppError("NOT_FOUND", "Document not found", 404)
89 else:
90 db.execute("UPDATE documents SET updated_at=%s WHERE document_id=%s", (now, document_id))
91 db.execute("""
92 INSERT INTO ingestion_jobs(task_id, document_id, status, stage, created_at, updated_at)
93 VALUES (%s, %s, 'queued', 'validated', %s, %s)
94 """, (task_id, document_id, now, now))
95 self.event(db, task_id)
96 # Duplicate uploads cannot grant access or undo revocation.
97 db.execute("INSERT INTO document_access VALUES (%s, %s) ON CONFLICT DO NOTHING",
98 (document_id, identity.user_id))
99 row = db.execute("""
100 SELECT d.document_id, d.filename, d.size_bytes, d.file_hash, d.source, d.version,
101 d.parser_version, j.task_id, j.status
102 FROM documents d JOIN ingestion_jobs j USING(document_id) WHERE d.document_id = %s
103 """, (document_id,)).fetchone()
104 return {**dict(row), "duplicate": duplicate}

106 def get_task(self, task_id: str, identity: Identity) -> dict:
107 with self.connection() as db:
108 db.execute("SET TRANSACTION ISOLATION LEVEL REPEATABLE READ")
109 row = db.execute("""
110 SELECT j.*, d.filename, d.file_hash, d.source, d.version, d.parser_version,
111 d.title, length(d.text) AS text_length
112 FROM ingestion_jobs j JOIN documents d USING(document_id)
113 JOIN document_access a USING(document_id)
114 WHERE j.task_id = %s AND d.tenant_id = %s AND a.user_id = %s AND d.active=1
115 """, (task_id, identity.tenant_id, identity.user_id)).fetchone()
116 if row is None:
117 raise AppError("NOT_FOUND", "Task not found", 404)
118 result = dict(row)
119 result.pop("claim_token")
120 result["events"] = [dict(event) for event in db.execute("""
121 SELECT status, stage, attempt, error_code, created_at FROM ingestion_events
122 WHERE task_id = %s ORDER BY event_id
123 """, (task_id,))]
124 return result

126 def retry(self, task_id: str, identity: Identity, max_attempts: int) -> dict:
127 with self.connection() as db:
128 allowed = db.execute("""SELECT 1 FROM ingestion_jobs j
129 JOIN documents d USING(document_id) JOIN document_access a USING(document_id)
130 WHERE j.task_id=%s AND d.tenant_id=%s AND a.user_id=%s AND d.active=1
131 FOR SHARE OF d, a""", (task_id, identity.tenant_id, identity.user_id)).fetchone()
132 if allowed is None:
133 raise AppError("NOT_FOUND", "Task not found", 404)
134 cursor = db.execute("""
135 UPDATE ingestion_jobs SET status = 'queued', stage = 'validated',
136 error_code = NULL, error_message = NULL, lease_until = NULL,
137 claim_token = NULL, updated_at = %s
138 WHERE task_id = %s AND status = 'failed' AND attempts < %s
139 """, (time.time(), task_id, max_attempts))
140 if not cursor.rowcount:
141 raise AppError("RETRY_NOT_ALLOWED", "Only failed tasks below the attempt limit can retry", 409)
142 self.event(db, task_id)
143 return self.get_task(task_id, identity)

145 def claim(self, lease_seconds: float, max_attempts: int, *, now: float | None = None) -> dict | None:
146 now = time.time() if now is None else now
147 with self.connection() as db:
148 db.execute("BEGIN")
149 row = db.execute("""
150 SELECT * FROM ingestion_jobs WHERE status = 'queued'
151 OR (status = 'processing' AND lease_until <= %s)
152 ORDER BY created_at, task_id LIMIT 1 FOR UPDATE SKIP LOCKED
153 """, (now,)).fetchone()
154 if row is None:
155 return None
156 if row["attempts"] >= max_attempts:
157 db.execute("""
158 UPDATE ingestion_jobs SET status = 'failed', error_code = 'ATTEMPT_LIMIT',
159 error_message = 'Processing attempt limit reached', claim_token = NULL,
160 lease_until = NULL, updated_at = %s WHERE task_id = %s
161 """, (now, row["task_id"]))
162 self.event(db, row["task_id"])
163 return None
164 token = str(uuid4())
165 db.execute("""
166 UPDATE ingestion_jobs SET status = 'processing', stage = 'parsing',
167 attempts = attempts + 1, claim_token = %s, lease_until = %s, updated_at = %s,
168 error_code = NULL, error_message = NULL WHERE task_id = %s
169 """, (token, now + lease_seconds, now, row["task_id"]))
170 self.event(db, row["task_id"])
171 document = dict(db.execute("SELECT * FROM documents WHERE document_id = %s",
172 (row["document_id"],)).fetchone())
173 return {**document, "task_id": row["task_id"], "claim_token": token}

176 with self.connection() as db:
177 cursor = db.execute("""

185 def complete(self, task: dict, parsed: ParsedDocument) -> bool:
186 with self.connection() as db:
187 cursor = db.execute("""
188 UPDATE ingestion_jobs SET status = 'succeeded', stage = 'completed',
189 lease_until = NULL, claim_token = NULL, updated_at = %s
190 WHERE task_id = %s AND claim_token = %s AND status = 'processing'
191 """, (time.time(), task["task_id"], task["claim_token"]))
192 if not cursor.rowcount:
193 return False
194 db.execute("""
195 UPDATE documents SET text = %s, title = %s, metadata = %s, version = COALESCE(version, %s)
196 WHERE document_id = %s
197 """, (parsed.text, parsed.title, json.dumps(parsed.metadata, ensure_ascii=False),
198 parsed.metadata.get("version", "unversioned"), task["document_id"]))
199 self.event(db, task["task_id"])
200 return True
