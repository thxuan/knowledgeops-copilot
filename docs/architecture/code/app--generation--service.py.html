24def validation_code(exc):
25 # Map only fixed validator messages; never log raw outputs or arbitrary fields.
26 known = {
27 "Answer must consist of the validated claims": "answer_claims_mismatch",
28 "Citation list must match claim citations": "citation_union_mismatch",
29 "Refusal must not assert evidence-backed claims": "refusal_has_claims",
30 "Every answer needs cited claims": "missing_cited_claims",
31 "Blank answer": "blank_answer",
32 "Citation outside provided context": "citation_outside_evidence",
33 }
34 if isinstance(exc, ValidationError):
35 for error in exc.errors(include_input=False, include_url=False):
36 if error["loc"] == ("answer",) and error["type"] == "string_too_short":
37 return "blank_answer"
38 code = known.get(str(error.get("ctx", {}).get("error", "")))
39 if code:
40 return code
41 return "schema_invalid"
42 return known.get(str(exc), "invalid_output")

45def schema_issues(exc):
46 if not isinstance(exc, ValidationError):
47 return []
48 # Allowlist paths/types; unknown extra field names can contain private text.
49 fields = {"answer", "citations", "insufficient_context", "claims", "text"}
50 types = {"missing", "extra_forbidden", "json_invalid", "json_type", "string_type",
51 "bool_type", "list_type", "model_type", "string_too_short", "string_too_long",
52 "too_short", "too_long", "value_error"}
53 result = []
54 for error in exc.errors(include_input=False, include_context=False, include_url=False)[:3]:
55 parts = [str(part) if part in fields else "[]" if isinstance(part, int) else "<extra>"
56 for part in error["loc"]]
57 result.append({"field": ".".join(parts) or "<root>",
58 "type": error["type"] if error["type"] in types else "other"})
59 return result
