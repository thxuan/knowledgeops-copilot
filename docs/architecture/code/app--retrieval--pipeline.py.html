52 async def prepare(self, query, identity, candidate_k, filters=None, mode="exact", ef_search=40,
53 include_lexical=True):
54 started = time.perf_counter()
55 result = await self.vectors.search(query, identity, candidate_k, mode, ef_search, filters)
56 vector_hits = [normalize(h) for h in result["hits"]]
57 lexical_started = time.perf_counter()
58 keyword_hits = []
59 if include_lexical:
60 allowed = await asyncio.to_thread(self.vectors.allowed_sets, identity, filters)
61 corpus = await asyncio.to_thread(self.vectors.vectors.lexical_candidates, allowed, identity.tenant_id)
62 keyword_hits = [normalize(h) for h in await asyncio.to_thread(bm25, query, corpus, candidate_k)]
63 vector_hits = await self.reauthorize(vector_hits, identity, filters)
64 keyword_hits = await self.reauthorize(keyword_hits, identity, filters)
65 fused = rrf({"vector": vector_hits, "keyword": keyword_hits}, self.settings.rrf_constant)
66 return {"query": query, "result": result, "vector_hits": vector_hits,
67 "keyword_hits": keyword_hits, "fused": fused[:candidate_k],
68 "fusion_union_count": len(fused), "candidate_k": candidate_k,
69 "lexical_ms": (time.perf_counter() - lexical_started) * 1000,
70 "prepare_ms": (time.perf_counter() - started) * 1000}

72 async def finish(self, prepared, identity, top_k, strategy, filters=None):
73 started = time.perf_counter()

75 hits = await self.reauthorize(hits, identity, filters)
76 candidates = [dict(h) for h in hits]
77 rerank = {"status": "disabled", "provider": self.settings.rerank.provider,
78 "duration_ms": 0.0, "input_count": 0, "total_tokens": 0, "estimated_cost": 0.0}
79 score_type = "cosine_similarity" if strategy == "vector" else "rrf"
80 if strategy == "hybrid_rerank":
81 if self.reranker is None:
82 raise AppError("RERANK_NOT_CONFIGURED", "Configure a rerank provider first", 503)
83 rerank.update(status="empty", input_count=len(hits))
84 if hits:
85 rerank_started = time.perf_counter()
86 try:
87 result = await asyncio.wait_for(self.reranker.rerank(prepared["query"],
88 [evidence_text(h) for h in hits]), self.settings.rerank.timeout_seconds)
89 validate_result(result, len(hits), self.settings.rerank.provider == "compatible")
90 hits = [{**hits[i], "score": score} for i, score in sorted(
91 result.results, key=lambda r: (-r[1], r[0]))]
92 price = self.settings.rerank.cost_per_million_tokens
93 cost = (0.0 if self.settings.rerank.provider == "fake" else
94 result.total_tokens / 1_000_000 * price
95 if result.total_tokens is not None and price is not None else None)
96 rerank.update(status="succeeded", total_tokens=result.total_tokens, estimated_cost=cost,
97 model_id=result.model_id, request_id=result.request_id)
98 score_type = "fake_bm25" if self.settings.rerank.provider == "fake" else "rerank_relevance"
99 except Exception:
100 rerank.update(status="failed_fallback", total_tokens=None, estimated_cost=None,
101 error_code="RERANK_FAILED")
102 rerank["duration_ms"] = round((time.perf_counter() - rerank_started) * 1000, 3)
103 # Recheck both final evidence and diagnostic payloads after external work.
104 authorized = {h["chunk_id"] for h in await self.reauthorize(hits + candidates, identity, filters)}
105 candidates = [h for h in candidates if h["chunk_id"] in authorized]
106 hits = [h for h in hits if h["chunk_id"] in authorized][:top_k]
107 base = prepared["result"]
108 duration = (base["duration_ms"] if strategy == "vector" else prepared["prepare_ms"])
109 duration += (time.perf_counter() - started) * 1000
110 return {**{k: base[k] for k in ("collection", "model_id", "dimensions", "mode", "run_id")},
111 "score_type": score_type, "duration_ms": round(duration, 3), "hits": hits,
112 "diagnostics": {"strategy": strategy, "original_query": prepared["query"],
113 "query_rewritten": False, "candidate_k": prepared["candidate_k"], "top_k": top_k,
114 "vector_count": len(prepared["vector_hits"]), "keyword_count": len(prepared["keyword_hits"]),
115 "fusion_union_count": prepared["fusion_union_count"], "candidate_count": len(candidates),
116 "final_count": len(hits), "candidates": candidates, "rerank": rerank,
117 "vector_ms": base["duration_ms"], "lexical_and_fusion_ms": round(prepared["lexical_ms"], 3),
118 "config": self.settings.public_config}}
