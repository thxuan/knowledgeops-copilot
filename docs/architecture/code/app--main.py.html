120 @app.middleware("http")
121 async def request_context(request: Request, call_next):
122 request_id = str(uuid4())
123 request.state.request_id = request_id
124 started = time.perf_counter()
125 try:
126 response = await call_next(request)
127 except Exception:
128 logger.exception(json.dumps({"event": "request.failed", "request_id": request_id}))
129 response = error_response("INTERNAL_ERROR", "Internal server error", request_id, 500)
130 response.headers["X-Request-ID"] = request_id
131 if request.url.path != "/health":
132 response.headers["Cache-Control"] = "no-store"
133 logger.info(json.dumps({
134 "event": "request.completed",
135 "request_id": request_id,
136 "method": request.method,
137 "path": request.url.path,
138 "user_id": getattr(request.state, "user_id", None),
139 "status_code": response.status_code,
140 "duration_ms": round((time.perf_counter() - started) * 1000, 2),
141 }, ensure_ascii=False))
142 return response
