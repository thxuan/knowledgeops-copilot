15def migrate(source_path, database_url):
16 source_path = Path(source_path).resolve(strict=True)
17 if not source_path.is_file():
18 raise ValueError("Source must be a SQLite file")
19 # One consistent source snapshot, never write or delete the original file.
20 with sqlite3.connect(source_path.as_uri() + "?mode=ro", uri=True) as source:
21 source.row_factory = sqlite3.Row
22 source.execute("BEGIN")
23 available = {row[0] for row in source.execute("SELECT name FROM sqlite_master WHERE type='table'")}
24 if "documents" not in available:
25 raise ValueError("Not a KnowledgeOps intake database")
26 tables = {table: [dict(row) for row in source.execute(f'SELECT * FROM "{table}"')]
27 if table in available else [] for table in TABLES}
28 for row in tables["documents"]:
29 row.setdefault("active", 1)
30 row.setdefault("revision", 1)
31 row.setdefault("updated_at", row["created_at"])
32 target = IngestionRepository(database_url)
33 initialize_core(target)
34 with target.connection() as db:
35 # Block concurrent writes during emptiness check and the all-or-nothing import.
36 db.execute(sql.SQL("LOCK TABLE {} IN ACCESS EXCLUSIVE MODE").format(
37 sql.SQL(",").join(map(sql.Identifier, TABLES))))
38 if any(db.execute(sql.SQL("SELECT EXISTS (SELECT 1 FROM {}) AS present").format(
39 sql.Identifier(table))).fetchone()["present"] for table in TABLES):
40 raise ValueError("Target core tables must be empty; use a separate database/schema for another import")
41 counts = restore_rows(db, tables)
42 for table, count in counts.items():
43 actual = db.execute(sql.SQL("SELECT count(*) AS value FROM {}").format(
44 sql.Identifier(table))).fetchone()["value"]
45 if actual != count:
46 raise ValueError("Imported row count mismatch")
47 return {"source": str(source_path), "source_sha256": hashlib.sha256(source_path.read_bytes()).hexdigest(),
48 "target": database_label(database_url), "counts": counts, "source_preserved": True}
