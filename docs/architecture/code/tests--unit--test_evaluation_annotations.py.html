56@pytest.mark.parametrize('damage', ['missing_source','missing_section','stale_hash','invalid_coordinate','wrong_section_range','duplicate_section'])
57def test_preflight_rejects_incomplete_or_stale_annotations(damage):
58 corpus, q = tiny_data()

64 elif damage == 'stale_hash':

67 anchors[0]['end_char'] = len(corpus['doc']['text']) + 1
68 elif damage == 'wrong_section_range':
