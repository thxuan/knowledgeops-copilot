124 def search(self, vector, allowed, tenant_id, top_k, mode="exact", ef_search=40, explain=False):
125 validate_result(EmbeddingResult(self.settings.model_id, [vector]), self.settings, 1)
126 if mode not in {"exact", "hnsw"} or not 1 <= top_k <= 50 or not 1 <= ef_search <= 1000:
127 raise ValueError("Invalid search configuration")
128 # OFFSET 0 preserves a parameterized LATERAL filter. The outer vector scan
129 # keeps distance order while validating each candidate against active/allowed sets.
130 query = sql.SQL("""SELECT c.payload, (e.embedding::vector({}) <=> %s::vector) AS distance
131 FROM ko_embeddings e
132 JOIN LATERAL (
133 SELECT c.payload FROM ko_chunks c JOIN ko_documents d USING(document_id)
134 JOIN ko_vector_heads h ON h.document_id=c.document_id
135 AND h.collection=e.collection AND h.chunk_set_id=c.chunk_set_id
136 JOIN jsonb_to_recordset(%s::jsonb) AS a(document_id text, chunk_set_id text)
137 ON a.document_id=c.document_id AND a.chunk_set_id=c.chunk_set_id
138 WHERE c.chunk_id=e.chunk_id AND d.tenant_id=%s
139 OFFSET 0
140 ) c ON true
141 WHERE e.collection={}
142 ORDER BY e.embedding::vector({}) <=> %s::vector LIMIT %s""").format(
143 sql.Literal(self.settings.dimensions), sql.Literal(self.settings.collection), sql.Literal(self.settings.dimensions))
144 params = (json.dumps(vector), Jsonb(allowed), tenant_id, json.dumps(vector), top_k)
145 with self.connection() as db:
146 self._check(db)
147 if mode == "exact":
148 db.execute("SET LOCAL enable_indexscan = off")
149 db.execute("SET LOCAL enable_bitmapscan = off")
150 else:
151 if db.execute("SELECT to_regclass(%s) AS name", (self.index_name,)).fetchone()["name"] is None:
152 raise AppError("HNSW_NOT_BUILT", "Build the HNSW index first", 409)
153 db.execute("SET LOCAL enable_seqscan = off")
154 db.execute("SELECT set_config('hnsw.ef_search', %s, true)", (str(ef_search),))
155 db.execute("SET LOCAL hnsw.iterative_scan = 'strict_order'")
156 if explain:
157 return db.execute(sql.SQL("EXPLAIN (ANALYZE, BUFFERS, FORMAT JSON) ") + query, params).fetchone()["QUERY PLAN"]
158 rows = db.execute(query, params).fetchall()
159 return [{**r["payload"], "distance": r["distance"], "score": 1 - r["distance"]} for r in rows]
