99def validate_fixture(intake, chunks, vectors, corpus):
100 source_ids = {"sample_data/knowledge_base/" + doc["filename"]: key for key, doc in corpus.items()}
101 with intake.connection() as db:
102 docs = [dict(r) for r in db.execute("""SELECT d.*,h.chunk_set_id FROM documents d
103 JOIN document_chunk_heads h USING(document_id) WHERE d.active=1""")]
104 acl = {d["document_id"]: sorted(r["user_id"] for r in db.execute(
105 "SELECT user_id FROM document_access WHERE document_id=%s", (d["document_id"],))) for d in docs}
106 if len(docs) != len(corpus) or {d["source"] for d in docs} != set(source_ids):
107 raise ValueError("Use the complete isolated 14-document sample index")
108 for d in docs:
109 doc = corpus[source_ids[d["source"]]]
110 if any(d[k] != doc[k] for k in ("version", "text", "parser_version")):
111 raise ValueError("Sample corpus differs from frozen sources")
112 if acl[d["document_id"]] != sorted(doc["allowed_user_ids"]):
113 raise ValueError("Sample ACL differs from reviewed fixture manifest")
114 snapshot = chunks.get(d["document_id"], LOCAL_IDENTITIES[doc["allowed_user_ids"][0]])
115 if not vectors.ready(d["document_id"], d["chunk_set_id"], snapshot["count"]):
116 raise ValueError("Sample vector index is incomplete or model signature differs")
117 return source_ids, docs

120def filter_checks(hits, identity, corpus, source_ids, filters=None):
121 filters = filters or {}
122 result = []
123 for hit in hits:
124 doc = corpus.get(source_ids.get(hit["source"]))
125 valid = bool(doc and identity.user_id in doc["allowed_user_ids"] and hit["version"] == doc["version"])
126 if filters.get("source") is not None:
127 valid = valid and hit["source"] == filters["source"]
128 if filters.get("document_ids") is not None:
129 valid = valid and hit["document_id"] in filters["document_ids"]
130 result.append(valid)
131 return result

134async def sampling(rows, settings, args, out):
135 if settings.generation.provider != "compatible" or not settings.generation.model_id.startswith("qwen-plus"):
136 write_json(out / "sampling.json", {"status": "skipped", "reason": "This experiment validates qwen-plus parameters only; Fake has no sampling behavior."})
137 return
138 ids = ["F005", "X001", "U001"]
139 selected = [r for qid in ids for r in rows if r["question_id"] == qid and r["generation_calls"]]
140 if len(selected) != 3:
141 raise ValueError("Sampling requires all three dev questions with recorded generation inputs")
142 write_json(out / "sampling-inputs.json", [
143 {"question_id": r["question_id"], "messages": r["generation_calls"][0]["messages"],
144 "messages_sha256": digest(r["generation_calls"][0]["messages"])} for r in selected])
145 configs = [("baseline", .7, .8), ("temperature_0", 0, .8), ("temperature_1", 1, .8),
146 ("top_p_04", .7, .4), ("top_p_1", .7, 1)]
147 tasks = [(r, name, t, p, n) for r in selected for name, t, p in configs for n in range(1, 4)]
148 random.Random(8).shuffle(tasks)
149 results = []
150 client = RecordingGeneration(make_generation_client(settings.generation))
151 for row, name, temperature, top_p, repeat in tasks:
152 gs = replace(settings.generation, temperature=temperature, top_p=top_p)
153 client.calls = []
154 messages = row["generation_calls"][0]["messages"]
155 item = {"question_id": row["question_id"], "configuration": name, "repeat": repeat,
156 "sent_temperature": temperature, "sent_top_p": top_p,
157 "messages_sha256": digest(messages), "status": "error",
158 "correctness": None, "citation_support": None, "faithfulness": None}
159 started = time.perf_counter()
160 try:
161 raw = await asyncio.wait_for(client.generate(messages, gs), gs.timeout_seconds)
162 parsed = parse_generation(raw)
163 evidence = json.loads(messages[-1]["content"])["evidence"]
164 if not set(parsed.citations) <= {h["chunk_id"] for h in evidence}:
165 raise ValueError("Citation outside provided context")
166 item.update(status="refused" if parsed.insufficient_context else "answered",
167 parsed_output=parsed.model_dump())
168 except (ValueError, AppError, TimeoutError) as exc:
169 item["error_code"] = exc.code if isinstance(exc, AppError) else validation_code(exc)
170 item.update(duration_ms=round((time.perf_counter() - started) * 1000, 3),
171 calls=client.calls, usage=generation_cost(client.calls, gs.provider, args.input_price, args.output_price))
172 results.append(item)
173 write_jsonl(out / "sampling-results.jsonl", results)
174 print(f"sampling {len(results)}/{len(tasks)} {row['question_id']} {name}", flush=True)
175 groups = []
176 for qid in ids:
177 for name, _, _ in configs:
178 group = [r for r in results if r["question_id"] == qid and r["configuration"] == name]
179 groups.append({"question_id": qid, "configuration": name, "attempts": len(group),
180 "distinct_raw_outputs": len({c["raw_output"] for r in group for c in r["calls"] if c["raw_output"] is not None}),
181 "error_count": sum(r["status"] == "error" for r in group)})
182 write_json(out / "sampling.json", {"status": "completed", "parameter_source": PARAMETER_SOURCE,
183 "parameter_check_date": "2026-10-03", "effective_parameters": "sent values recorded; provider does not echo effective values",
184 "model_version_limit": "Use a dated model ID for pinned reproducibility; qwen-plus is an unpinned alias.",
185 "scope": "dev only, fixed first-attempt inputs, one generation per attempt, no repair; semantic review pending",
186 "attempt_count": len(results), "groups": groups})

253async def run_evaluation(args, settings, corpus, questions, anchors, out, manifest):
254 run_id = manifest["run_id"]
255 # Copy core tables into an isolated PostgreSQL schema; preserve a portable JSON snapshot.
256 intake, schema = evaluation_snapshot(args.database_url or settings.database_url, out)
257 manifest["snapshot_schema"] = schema
258 intake.initialize()
259 chunks = ChunkRepository(intake)
260 chunks.initialize()
261 conversations = ConversationRepository(intake)
262 conversations.initialize()
263 vectors = VectorRepository(args.database_url or settings.database_url, settings.embedding)
264 vectors.initialize()
265 source_ids, docs = validate_fixture(intake, chunks, vectors, corpus)
266 manifest["corpus"] = [{k: d[k] for k in ("document_id", "source", "version", "chunk_set_id")} for d in docs]
267 write_json(out / "manifest.json", manifest)
268 service = RetrievalService(VectorService(chunks, vectors, make_embedding_client(settings.embedding),
269 settings.embedding), settings.retrieval)
270 client = RecordingGeneration(make_generation_client(settings.generation))
271 rows = []
272 for q in questions:
273 identity = LOCAL_IDENTITIES[q["allowed_user_ids"][0]]
274 row = {"question_id": q["question_id"], "question": q["question"], "user_id": identity.user_id,
275 "question_type": q["question_type"], "answerable": q["answerable"],
276 "expected_answer_points": q["expected_answer_points"], "expected_behavior": q["expected_behavior"],
277 "status": "error", "retrieval": {}, "hits": [], "citation_checks": [], "filter_checks": [],
278 "generation_calls": [], "failure_category": None}
279 client.calls = []
280 started = time.perf_counter()
281 try:
282 result = await service.retrieve(q["question"], identity, top_k=args.top_k)
283 row["hits"] = result["hits"]
284 row["retrieval_duration_ms"] = result["duration_ms"]
285 row["embedding_batches"] = vectors.batch_events(result["run_id"])
286 row["retrieval"] = measure(q, result["hits"], anchors, source_ids)
287 row["filter_checks"] = filter_checks(result["hits"], identity, corpus, source_ids)
288 response = await ChatService(conversations, FixedRetrieval(result), client, settings.generation).run(
289 ChatRequest(message=q["question"], top_k=args.top_k), identity)
290 row["response"] = response
291 row["status"] = response["status"]
292 context = build_context(q["question"], [], {}, result["hits"], settings.generation)
293 row["input_evidence"] = context.evidence
294 ids = {h["chunk_id"] for h in context.evidence}
295 row["citation_checks"] = [c["chunk_id"] in ids for c in response["citations"]]
296 if row["status"] == "rejected":
297 row["failure_category"] = "generation_validation"
298 elif row["status"] == "refused" and q["answerable"]:
299 row["failure_category"] = ("retrieval_or_evidence_missing" if row["retrieval"]["source_recall_at_k"] < 1
300 else "context_or_generation_review_needed")
301 elif row["status"] == "answered" and not q["answerable"]:
302 row["failure_category"] = "unanswerable_answered_review_needed"
303 except AppError as exc:
304 row["error_code"] = exc.code
305 row["failure_category"] = "runtime_error"
306 row["generation_calls"] = client.calls
307 row["generation_usage"] = generation_cost(client.calls, settings.generation.provider, args.input_price, args.output_price)
308 row["duration_ms"] = round((time.perf_counter() - started) * 1000, 3)
309 rows.append(row)
310 write_jsonl(out / "results.jsonl", rows)
311 print(f"{len(rows)}/{len(questions)} {q['question_id']} {row['status']}", flush=True)
312 summary = summarize(rows)
313 usage_rows = [r["generation_usage"] for r in rows]
314 summary["generation_usage"] = {
315 key: sum(u[key] for u in usage_rows) if all(u[key] is not None for u in usage_rows) else None
316 for key in ("prompt_tokens", "completion_tokens", "estimated_cost")}
317 summary["generation_usage"]["complete_usage_samples"] = sum(u["prompt_tokens"] is not None for u in usage_rows)
318 summary["currency"] = args.currency
319 batch_rows = [r.get("embedding_batches", []) for r in rows]
320 summary["embedding_usage"] = {
321 "total_tokens": sum(e["total_tokens"] for events in batch_rows for e in events)
322 if all(events and all(e["total_tokens"] is not None for e in events) for events in batch_rows) else None,
323 "estimated_cost": sum(e["estimated_cost"] for events in batch_rows for e in events)
324 if all(events and all(e["estimated_cost"] is not None for e in events) for events in batch_rows) else None,
325 "scope": "query embedding only; prior corpus indexing cost excluded"}
326
327 write_json(out / "summary.json", summary)
328 write_jsonl(out / "review-template.jsonl", [
329 {"run_id": run_id, "question_id": r["question_id"], "sample_sha256": digest(r),
330 "reviewer": "", "correctness": "", "citation_support": "", "faithfulness": "",
331 "failure_stage": "", "notes": ""} for r in rows])
332 # Dedicated filters independent of gold labels: source restriction and inaccessible engineering document.
333 employee = LOCAL_IDENTITIES["u_employee_001"]
334 filters = [{"source": "sample_data/knowledge_base/" + corpus["KB-NET-001"]["filename"]},
335 {"document_ids": [d["document_id"] for d in docs if source_ids[d["source"]] == "KB-ENG-001"]}]
336 probes = []
337 for f in filters:
338 result = await service.retrieve("VPN access", employee, top_k=args.top_k, filters=f)
339 probes.append({"user_id": employee.user_id, "filters": f, "hits": result["hits"],
340 "checks": filter_checks(result["hits"], employee, corpus, source_ids, f)})
341 write_json(out / "filter-probes.json", probes)
342 if args.sampling:
343 await sampling(rows, settings, args, out)
344 manifest.update(status="completed", sample_count=len(rows), semantic_review="pending")
345 write_json(out / "manifest.json", manifest)
346 print(json.dumps({"run_directory": str(out), "summary": summary}, ensure_ascii=False, indent=2))
