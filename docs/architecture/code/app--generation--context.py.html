36def build_context(question, history, summary, hits, settings):
37 payload = {"question": question, "history": [], "summary": [], "evidence": []}
38 def messages():
39 return [{"role": "system", "content": SYSTEM_PROMPT},
40 {"role": "user", "content": dump(payload)}]
41 limit = settings.input_limit - settings.output_limit
42 if estimate_tokens(messages()) > limit:
43 raise AppError("CONTEXT_BUDGET_EXCEEDED", "System rules and current question exceed context budget", 422)
44 selected = []
45 history_limit = min(settings.history_budget, max(0, limit - estimate_tokens(messages())) // 3)
46 for row in reversed(history):
47 if row["role"] != "user":
48 continue
49 item = {"message_id": row["message_id"], "text": row["content"]}
50 if len(dump([item, *selected]).encode("utf-8")) > history_limit:
51 break # Never skip a recent correction and resurrect older entities.
52 selected.insert(0, item)
53 payload["history"] = selected
54 evidence = []
55 for hit in hits:
56 item = {"chunk_id": hit["chunk_id"], "source": hit["source"],
57 "version": hit["version"], "text": hit["text"]}
58 payload["evidence"].append(item)
59 if estimate_tokens(messages()) > limit:
60 payload["evidence"].pop()
61 else:
62 evidence.append(hit)
63 # Old exact user records; original IDs enable lookup. Optional if space is short.
64 summary_records = summary.get("records", []) if selected or not history else []
65 for item in reversed(summary_records):
66 payload["summary"].insert(0, item)
67 if estimate_tokens(messages()) > limit:
68 payload["summary"].pop(0)
69 break
70 recent_ids = [i["message_id"] for i in selected]
71 summary_ids = [i["message_id"] for i in payload["summary"]]

77def retrieval_query(question, recent_users):
78 # Explicit current identifiers take priority. For a pronoun query, select the
79 # latest code-bearing topic within the recent three user records, not all topics.
80 if ERROR_CODE.search(question):
81 return question
82 if recent_users and any(word in question for word in ("它", "这个", "那个", "上述", "刚才", "多久", "然后")):
83 window = recent_users[-3:]
84 anchor = next((m for m in reversed(window) if ERROR_CODE.search(m["content"])), window[-1])
85 previous = anchor["content"]
86 query = question + "\n前一轮问题：" + previous
87 # Preserve full correction records; never truncate away the new identifier.
88 if len(previous) <= 2048 and len(query) <= 4096:
89 return query
90 return question
