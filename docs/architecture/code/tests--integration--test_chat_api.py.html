24class LocalRetrieval:
25 """Test double for ranking only; citations and ACLs use the real repository."""
26 def __init__(self, app, hits):
27 self.app, self.hits, self.queries = app, hits, []
28 async def retrieve(self, query, identity, **kwargs):
29 self.queries.append(query)
30 hits = []
31 for hit in self.hits:
32 try:
33 self.app.state.document_repository.citation(hit["chunk_id"], hit["citation"]["chunk_set_id"], identity)
34 except AppError:
35 continue
36 ids = kwargs.get("filters", {}).get("document_ids")
37 if ids is None or hit["document_id"] in ids:
38 hits.append(hit)
39 return {"hits": hits[:kwargs.get("top_k", 5)]}

41def seed(app, client, text, source):

43 files={"file": (source + ".md", ("# 支持\n\n" + text).encode())},
44 data={"source": source, "version": "v1"}).json()
45 app.state.ingestion_worker.process_next()

196 class Barrier:
197 def __init__(self):
198 self.entered = 0
199 self.ready = asyncio.Event()
200 async def generate(self, messages, settings):
201 self.entered += 1
202 if self.entered == 2:
203 self.ready.set()
204 await self.ready.wait()
205 return await FakeGenerationClient().generate(messages, settings)
206 async def run():
207 app.state.chat.client = Barrier()
