77 async def embed(self, texts):
78 vectors = []
79 for text in texts:
80 tokens = re.findall(r"[a-z0-9]+(?:[-_.][a-z0-9]+)*", text.lower())
81 for run in re.findall(r"[\u4e00-\u9fff]+", text):
82 tokens.extend(run[i:i + 2] for i in range(max(1, len(run) - 1)))
83 vector = [0.0] * self.settings.dimensions
84 for token in tokens or ["<empty>"]:
85 digest = hashlib.sha256(token.encode()).digest()
86 vector[int.from_bytes(digest[:4], "big") % len(vector)] += 1 if digest[4] % 2 else -1
87 norm = math.hypot(*vector)
88 if norm == 0:
89 vector[0], norm = 1.0, 1.0
90 vectors.append([x / norm for x in vector])
91 return EmbeddingResult(self.settings.model_id, vectors, 0)

97 async def embed(self, texts):

99 async with httpx.AsyncClient(timeout=self.settings.timeout_seconds) as client:
100 response = await client.post(self.settings.base_url.rstrip("/") + "/embeddings",
101 headers={"Authorization": "Bearer " + self.settings.api_key},
102 json={"model": self.settings.model_id, "input": texts, "encoding_format": "float"})
103 if response.status_code == 429 or response.status_code >= 500:
104 raise EmbeddingTransientError("Provider unavailable")
105 if response.status_code != 200:
106 raise EmbeddingError("Provider rejected request")
107 body = response.json()
108 data = sorted(body["data"], key=lambda row: row["index"])
109 if [row["index"] for row in data] != list(range(len(texts))):
110 raise EmbeddingError("Invalid embedding response indices")
111 return EmbeddingResult(body["model"], [row["embedding"] for row in data],
