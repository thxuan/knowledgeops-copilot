67def _scan_lines(text: str):
68 """Recognize ATX headings and blank boundaries outside backtick/tilde fences."""
69 offset, fence = 0, None
70 for line in text.splitlines(keepends=True):
71 marker = re.match(r"^ {0,3}(`{3,}|~{3,})(.*)$", line.rstrip("\n"))
72 inside = fence is not None
73 if marker:
74 run, suffix = marker.groups()
75 if fence is None:
76 fence = (run[0], len(run))
77 elif run[0] == fence[0] and len(run) >= fence[1] and not suffix.strip():
78 fence = None
79 heading = None
80 if not inside and not marker:
81 heading = re.match(r"^ {0,3}(#{1,6})[ \t]+(.+?)[ \t]*\n?$", line)
82 yield offset, line, heading, not inside and not marker and not line.strip()
83 offset += len(line)

86def markdown_sections(text: str, markdown: bool = True) -> list[Section]:
87 headings, path = [], []
88 if markdown:
89 for offset, _, match, _ in _scan_lines(text):
90 if match:
91 level = len(match.group(1))
92 label = re.sub(r"[ \t]+#+[ \t]*$", "", match.group(2)).strip()
93 path = [(depth, title) for depth, title in path if depth < level]
94 path.append((level, label))
95 headings.append((offset, level, tuple(title for _, title in path)))
96 if not headings:
97 return [Section(0, len(text), (), 0, len(text))]
98 sections = []
99 if headings[0][0] > 0:
100 sections.append(Section(0, headings[0][0], (), 0, headings[0][0]))
101 # Next heading of equal/lower depth closes a subtree; compute in linear time.
102 subtree_ends, stack = [len(text)] * len(headings), []
103 for index in range(len(headings) - 1, -1, -1):
104 while stack and headings[stack[-1]][1] > headings[index][1]:
105 stack.pop()
106 if stack:
107 subtree_ends[index] = headings[stack[-1]][0]
108 stack.append(index)
109 for index, (start, level, heading_path) in enumerate(headings):
110 end = headings[index + 1][0] if index + 1 < len(headings) else len(text)
111 subtree_end = subtree_ends[index]
112 sections.append(Section(start, end, heading_path, level, subtree_end))
113 return sections

116def chunk_document(document_id: str, text: str, *, title: str, source: str,
117 version: str, parser_version: str, config: ChunkConfig,
118 markdown: bool = True, generation: int = 1) -> list[Chunk]:
119 if not text.strip():
120 raise ValueError("Cannot chunk empty text")
121 sections = markdown_sections(text, markdown)
122 section_starts = [s.start for s in sections]
123 revision = text_revision(text)
124 line_starts = [0] + [i + 1 for i, char in enumerate(text) if char == "\n"]
125 ranges = []
126 scopes = [(0, len(text))] if config.strategy == "fixed" else [(s.start, s.end) for s in sections]
127 boundaries = [offset + len(line) for offset, line, _, blank in _scan_lines(text) if blank]
128 for scope_start, scope_end in scopes:
129 start = scope_start
130 while start < scope_end:
131 limit = min(start + config.size, scope_end)
132 end, reason = limit, "fixed"
133 if config.strategy == "structure":
134 candidate_index = bisect_right(boundaries, limit) - 1
135 candidate = boundaries[candidate_index] if candidate_index >= 0 else start
136 if limit == scope_end:
137 reason = "boundary"
138 elif candidate > start + config.overlap:
139 end, reason = candidate, "boundary"
140 else:
141 reason = "hard_limit"
142 ranges.append((start, end, reason))
143 if end == scope_end:
144 break
145 start = end - config.overlap
146 chunks = []
147 for index, (start, end, reason) in enumerate(ranges):
148 first = max(0, bisect_right(section_starts, start) - 1)
149 stop = bisect_left(section_starts, end)
150 paths = tuple(dict.fromkeys(s.heading_path for s in sections[first:stop]))
151 key = json.dumps([document_id, revision, parser_version, config.config_id, source, version, title, generation, start, end])
152 chunks.append(Chunk(
153 document_id=document_id, chunk_id=hashlib.sha256(key.encode()).hexdigest(), ordinal=index,
154 text=text[start:end], title=title, heading_paths=paths, start_char=start, end_char=end,
155 start_line=bisect_right(line_starts, start), end_line=bisect_right(line_starts, end - 1),
156 source=source, version=version, parser_version=parser_version, text_hash=revision,
157 config_id=config.config_id, split_reason=reason,
158 ))
159 return [replace(c, previous_chunk_id=chunks[i - 1].chunk_id if i else None,
160 next_chunk_id=chunks[i + 1].chunk_id if i + 1 < len(chunks) else None)
161 for i, c in enumerate(chunks)]
